<script>
    // ==================== 真实数据接口 ====================
    async function fetchRealDltHistory(pageCount = 2) {
        const allIssues = [];

        for (let page = 1; page <= pageCount; page++) {
            const url =
                "https://webapi.sporttery.cn/gateway/lottery/getHistoryPageListV1.qry" +
                "?gameNo=85" +
                "&provinceId=0" +
                "&pageSize=30" +
                "&isVerify=1" +
                "&pageNo=" + page;

            try {
                const res = await fetch(url, {
                    headers: { "Accept": "application/json" }
                });

                if (!res.ok) throw new Error("状态码：" + res.status);

                const json = await res.json();
                const list = json?.value?.list || json?.data?.list || json?.list || [];

                for (const item of list) {
                    const lotteryResult = item.lotteryDrawResult || "";
                    const parts = lotteryResult.split(":");
                    if (parts.length !== 2) continue;

                    const front = parts[0].trim().split(/\s+/).map(Number);
                    const back = parts[1].trim().split(/\s+/).map(Number);

                    if (front.length !== 5 || back.length !== 2) continue;

                    allIssues.push({
                        expect: item.lotteryDrawIssue || "",
                        date: item.lotteryDrawTime || "",
                        red: front,
                        blue: back
                    });
                }
            } catch (err) {
                console.error("第" + page + "页数据加载失败：", err);
            }
        }

        return allIssues;
    }

    function generateMockData(count = 50) {
        const data = [];
        for (let i = 0; i < count; i++) {
            let r = Array.from({length:35}, (_,k)=>k+1).sort(()=>0.5-Math.random()).slice(0,5);
            let b = Array.from({length:12}, (_,k)=>k+1).sort(()=>0.5-Math.random()).slice(0,2);
            data.push({ red: r, blue: b });
        }
        return data;
    }

    // ==================== 数据分析 ====================
    let redStats = {};
    let blueStats = {};
    let historyData = [];

    function analyzeData(data) {
        historyData = data;
        for (let i = 1; i <= 35; i++) redStats[i] = { count: 0, omission: 0 };
        for (let i = 1; i <= 12; i++) blueStats[i] = { count: 0, omission: 0 };

        data.forEach(issue => {
            issue.red.forEach(n => { if (redStats[n]) redStats[n].count++; });
            issue.blue.forEach(n => { if (blueStats[n]) blueStats[n].count++; });
        });

        for (let i = 1; i <= 35; i++) {
            let index = data.findIndex(d => d.red.includes(i));
            redStats[i].omission = index === -1 ? data.length : index;
        }
        for (let i = 1; i <= 12; i++) {
            let index = data.findIndex(d => d.blue.includes(i));
            blueStats[i].omission = index === -1 ? data.length : index;
        }
    }

    // ==================== 渲染面板 ====================
    function renderGrid() {
        const redContainer = document.getElementById('redGrid');
        const blueContainer = document.getElementById('blueGrid');
        redContainer.innerHTML = '';
        blueContainer.innerHTML = '';

        const redAvgCount = historyData.length * (5 / 35);
        const blueAvgCount = historyData.length * (2 / 12);

        for (let i = 1; i <= 35; i++) {
            let stat = redStats[i];
            let className = 'grid-item';
            if (stat.count >= redAvgCount + 2) className += ' hot';
            else if (stat.count <= redAvgCount - 2) className += ' cold';

            let div = document.createElement('div');
            div.className = className;
            div.innerHTML = `${i < 10 ? '0' + i : i}<span class="omission-tag">遗${stat.omission}</span>`;
            redContainer.appendChild(div);
        }

        for (let i = 1; i <= 12; i++) {
            let stat = blueStats[i];
            let className = 'grid-item';
            if (stat.count >= blueAvgCount + 1) className += ' hot';
            else if (stat.count <= blueAvgCount - 1) className += ' cold';

            let div = document.createElement('div');
            div.className = className;
            div.innerHTML = `${i < 10 ? '0' + i : i}<span class="omission-tag">遗${stat.omission}</span>`;
            blueContainer.appendChild(div);
        }
    }

    // ==================== 智能选号 ====================
    let currentMode = 'random';

    function setMode(mode) {
        currentMode = mode;
        document.querySelectorAll('.mode-btn').forEach(btn => btn.classList.remove('active'));
        event.target.classList.add('active');
    }

    function weightedSelect(totalNum, selectCount, statsObj, mode) {
        let pool = [];
        for (let i = 1; i <= totalNum; i++) pool.push(i);

        if (mode === 'random') {
            return pool.sort(() => 0.5 - Math.random()).slice(0, selectCount);
        }

        let weights = pool.map(num => {
            let s = statsObj[num];
            if (!s) return 1;
            if (mode === 'hot') return s.count + 1;
            if (mode === 'cold') return s.omission + 1;
            return 1;
        });

        let selected = new Set();
        while (selected.size < selectCount) {
            let totalWeight = 0;
            pool.forEach((num, idx) => {
                if (!selected.has(num)) totalWeight += weights[idx];
            });

            let random = Math.random() * totalWeight;
            let currentSum = 0;

            for (let i = 0; i < pool.length; i++) {
                if (selected.has(pool[i])) continue;
                currentSum += weights[i];
                if (random <= currentSum) {
                    selected.add(pool[i]);
                    break;
                }
            }
        }
        return Array.from(selected);
    }

    function generateNumbers() {
        const display = document.getElementById('resultDisplay');
        const btn = document.getElementById('generateBtn');
        btn.disabled = true;
        display.innerHTML = '<span style="color:#e03131">正在分析数据...</span>';

        setTimeout(() => {
            let reds = weightedSelect(35, 5, redStats, currentMode);
            reds.sort((a, b) => a - b);

            let blues = weightedSelect(12, 2, blueStats, currentMode);
            blues.sort((a, b) => a - b);

            let html = '';
            reds.forEach(n => html += `<div class="lotto-ball ball-red">${n < 10 ? '0' + n : n}</div>`);
            blues.forEach(n => html += `<div class="lotto-ball ball-blue">${n < 10 ? '0' + n : n}</div>`);
            display.innerHTML = html;

            addToHistory(reds, blues);
            btn.disabled = false;
        }, 500);
    }

    function addToHistory(reds, blues) {
        const list = document.getElementById('historyList');
        const time = new Date().toLocaleTimeString();
        let rHtml = reds.map(n => `<span class="r">${n < 10 ? '0' + n : n}</span>`).join(' ');
        let bHtml = blues.map(n => `<span class="b">${n < 10 ? '0' + n : n}</span>`).join(' ');

        let item = document.createElement('div');
        item.className = 'history-item';
        item.innerHTML = `
            <span style="color:#999; font-size:12px;">${time}</span>
            <div class="history-nums">${rHtml} + ${bHtml}</div>
        `;
        list.prepend(item);
    }

    function clearHistory() {
        document.getElementById('historyList').innerHTML = '';
    }

    // ==================== 初始化 ====================
    async function initApp() {
        const display = document.getElementById('resultDisplay');
        const btn = document.getElementById('generateBtn');
        const tip = document.getElementById('dataSourceTip');

        try {
            const realData = await fetchRealDltHistory(2);

            if (realData.length > 0) {
                analyzeData(realData);
                renderGrid();
                display.innerHTML = '<span style="color:#999">数据加载完成，点击按钮开始选号</span>';
                tip.textContent = `已加载 ${realData.length} 期真实开奖数据`;
                btn.disabled = false;
            } else {
                throw new Error("数据为空");
            }
        } catch (e) {
            console.error("真实数据加载失败，切换为模拟数据：", e);
            const mockData = generateMockData(50);
            analyzeData(mockData);
            renderGrid();
            display.innerHTML = '<span style="color:#e03131">真实数据加载失败，已使用模拟数据</span>';
            tip.textContent = "当前为模拟数据（接口请求受限）";
            btn.disabled = false;
        }
    }

    initApp();
</script>
